function doLogin(){
  const user = document.getElementById('loginUser').value.trim();
  const pass = document.getElementById('loginPass').value.trim();
  if(!user || !pass){
    Swal.fire({icon:'warning',title:'กรุณากรอกข้อมูลให้ครบ',text:'ป้อนชื่อผู้ใช้และรหัสผ่าน'});
    return;
  }
  if(user==='teacher' && pass==='666222'){
    window.currentUser = {role:'teacher'};
    document.getElementById('loginPage').style.display = 'none';
    document.getElementById('teacherPage').style.display = 'block';
    document.getElementById('studentPage').style.display = 'none';
    Swal.fire({icon:'success',title:'เข้าสู่ระบบสำเร็จ',text:'ยินดีต้อนรับคุณครูเข้าสู่ระบบจัดเก็บข้อมูลเรียลไทม์',timer:1500,showConfirmButton:false});

    if(stu){
      window.currentUser = {role:'student', data:stu};
      document.getElementById('loginPage').style.display = 'none';
      document.getElementById('teacherPage').style.display = 'none';
      document.getElementById('studentPage').style.display = 'block';
      renderStudentPage();
      Swal.fire({icon:'success',title:'เข้าสู่ระบบสำเร็จ',text:'ยินดีต้อนรับนักเรียน ' + stu.name,timer:1500,showConfirmButton:false});
    } else {
      Swal.fire({icon:'error',title:'ไม่พบข้อมูล',text:'ไม่พบรหัสนักเรียนนี้ในระบบ กรุณาแจ้งคุณครูผู้ควบคุมระบบเพื่อลงทะเบียน'});
    }
  } else {

  }
}

function renderClassrooms(){
  const el = document.getElementById('classroomList');
  if(!el) return;
  el.innerHTML = window.classrooms.map((r,i)=>`
    <div class="flex justify-between items-center bg-white/70 px-4 py-3 rounded-2xl border border-emerald-50">

      <button onclick="removeClassroom(${i})" class="text-red-500 text-sm font-bold hover:underline">ลบห้อง</button>
    </div>
  `).join('');
  updateRoomSelects();
}
async function addClassroom(){
  const v = document.getElementById('newRoom').value.trim();
  if(!v) return;
  if(window.classrooms.includes(v)) {
    Swal.fire({icon:'warning',title:'มีห้องเรียนนี้อยู่แล้ว'});
    return;
  }
  window.classrooms.push(v);
  document.getElementById('newRoom').value='';
  renderClassrooms();
  saveLocalStorage();
  if(window.updateClassroomsCloud) await window.updateClassroomsCloud();
  Swal.fire({icon:'success',title:'เพิ่มห้องเรียนสำเร็จ',timer:1000,showConfirmButton:false});
}
async function removeClassroom(i){
  window.classrooms.splice(i,1);
  renderClassrooms();
  saveLocalStorage();
  if(window.updateClassroomsCloud) await window.updateClassroomsCloud();
  Swal.fire({icon:'info',title:'ลบห้องเรียนเรียบร้อย',timer:1000,showConfirmButton:false});
}
function updateRoomSelects(){
  const opts = window.classrooms.map(r=>`<option value="${r}">ม.${r}</option>`).join('');
  const sel = document.getElementById('stuRoom');
  if(sel) sel.innerHTML = `<option value="">-- เลือกห้อง --</option>`+opts;
  
  const roomFilter = document.getElementById('roomFilter');
  if(roomFilter) roomFilter.innerHTML = `<option value="">📚 ทั้งหมด</option>` + opts;
}

async function importStudentList(){
  const txt = document.getElementById('importStudents').value.trim();
  if(!txt){Swal.fire({icon:'warning',title:'กรุณาใส่ข้อมูลนักเรียน'});return;}
  const lines = txt.split('\n').filter(l=>l.trim());
  let count=0;
  lines.forEach(l=>{
    const parts = l.split(',');
    if(parts.length>=3){
      const id=parts[0].trim(), name=parts[1].trim(), room=parts[2].trim();
      if(!window.students.find(s=>s.id===id)){
        window.students.push({id,name,room,avatar:""});
        count++;
      }
    }
  });
  document.getElementById('importStudents').value='';
  renderStudentsByRoom();
  saveLocalStorage();
  if(window.updateStudentsCloud) await window.updateStudentsCloud();
  Swal.fire({icon:'success',title:`นำเข้าสำเร็จจำนวน ${count} คน`,timer:1500,showConfirmButton:false});
}
function importStudentExcel(e){
  const file = e.target.files[0];
  if(!file) return;
  const reader = new FileReader();
  reader.onload = async function(ev){
    try{
      const data = new Uint8Array(ev.target.result);
      const workbook = XLSX.read(data, {type:'array'});
      const sheet = workbook.Sheets[workbook.SheetNames[0]];
      const rows = XLSX.utils.sheet_to_json(sheet, {header:1});
      let count=0;
      rows.forEach((row,idx)=>{
        if(idx===0) return; // Skip Header
        if(row.length>=3){
          const id=String(row[0]).trim(), name=String(row[1]).trim(), room=String(row[2]).trim();
          if(id && name && room && !window.students.find(s=>s.id===id)){
            window.students.push({id,name,room,avatar:""});
            count++;
          }
        }
      });
      renderStudentsByRoom();
      saveLocalStorage();
      if(window.updateStudentsCloud) await window.updateStudentsCloud();
      Swal.fire({icon:'success',title:`Import Excel สำเร็จ ${count} คน`,timer:1500,showConfirmButton:false});
    }catch(err){
      Swal.fire({icon:'error',title:'นำเข้าไฟล์ Excel ไม่สำเร็จ',text:err.message});
    }
  };
  reader.readAsArrayBuffer(file);
  e.target.value='';
}
function renderStudentsByRoom(){
  const search = (document.getElementById('searchStudent')?.value||'').toLowerCase();
  const selectedRoom = document.getElementById('roomFilter')?.value || '';
  let filtered = window.students.filter(s=>s.id.includes(search)||s.name.toLowerCase().includes(search));
  if(selectedRoom) filtered = filtered.filter(s=>s.room===selectedRoom);
  
  const el = document.getElementById('studentTable');
  if(!el) return;
  if(filtered.length===0){el.innerHTML='<p class="text-gray-400 text-center py-4">ไม่พบข้อมูลนักเรียน</p>';return;}
  
  const grouped = {};
  filtered.forEach(s=>{
    if(!grouped[s.room]) grouped[s.room] = [];
    grouped[s.room].push(s);
  });
  
  let html = '';
  Object.keys(grouped).sort().forEach(room=>{
    html += `
    <div class="mb-4 bg-white/50 p-3 rounded-2xl border border-gray-100">
      <div class="bg-emerald-600 px-4 py-2 rounded-xl font-bold text-white mb-2 text-sm flex justify-between items-center">
        <span>📚 มัธยมศึกษาปีที่ ${room}</span>
        <span class="text-xs">จำนวน ${grouped[room].length} คน</span>
      </div>
      <table class="w-full text-sm">
        <thead>
          <tr class="text-emerald-900 border-b border-emerald-100 text-left">
            <th class="px-3 py-2 w-1/4">รหัสนักเรียน</th>
            <th class="px-3 py-2 w-1/2">ชื่อ-สกุล</th>
            <th class="px-3 py-2 text-center w-1/4">จัดการ</th>
          </tr>
        </thead>
        <tbody>
          ${grouped[room].map(s=>`
            <tr class="border-b border-gray-100 hover:bg-emerald-50/40 transition-colors">
              <td class="px-3 py-2 font-mono font-bold text-gray-700">${s.id}</td>
              <td class="px-3 py-2 font-semibold text-gray-800">${s.name}</td>
              <td class="px-3 py-2 text-center">
                <button onclick="editStudent('${s.id}')" class="text-blue-600 text-xs font-bold mr-3 hover:underline">แก้ไข</button>
                <button onclick="deleteStudent('${s.id}')" class="text-red-500 text-xs font-bold hover:underline">ลบ</button>
              </td>
            </tr>
          `).join('')}
        </tbody>
      </table>
    </div>`;
  });
  el.innerHTML = html;
}
function deleteStudent(id){
  Swal.fire({
    title:'ลบรายชื่อนักเรียน?',
    text:'คุณแน่ใจว่าต้องการลบนักเรียนคนนี้ออกจากระบบคลาวด์หรือไม่?',
    icon:'warning',
    showCancelButton:true,
    confirmButtonText:'ยืนยันการลบ',
    cancelButtonText:'ยกเลิก',
    confirmButtonColor:'#dc2626'
  }).then(async res=>{
    if(res.isConfirmed){
      window.students = window.students.filter(s=>s.id!==id);
      renderStudentsByRoom();
      saveLocalStorage();
      if(window.updateStudentsCloud) await window.updateStudentsCloud();
      Swal.fire('สำเร็จ', 'ลบนักเรียนเสร็จสิ้น', 'success');
    }
  });
}
function editStudent(id){
  const s = window.students.find(x=>x.id===id);
  if(!s) return;
  Swal.fire({
    title:'แก้ไขข้อมูลนักเรียน',
    html:`
      <div class="text-left mb-2 text-sm font-semibold text-gray-600">ชื่อ-นามสกุล</div>
      <input id="swal-name" class="swal2-input !mt-1 !mb-4" value="${s.name}" placeholder="ชื่อ-สกุล">
      <div class="text-left mb-2 text-sm font-semibold text-gray-600">ห้องเรียน (ระบุเช่น 6/1)</div>
      <input id="swal-room" class="swal2-input !mt-1" value="${s.room}" placeholder="ห้อง">`,
    showCancelButton:true,
    confirmButtonText:'บันทึก',
    cancelButtonText:'ยกเลิก',
    preConfirm:()=>{
      const name = document.getElementById('swal-name').value.trim();
      const room = document.getElementById('swal-room').value.trim();
      if(!name || !room) {
        Swal.showValidationMessage('กรุณากรอกข้อมูลให้ครบถ้วน');
      }
      return {name, room};
    }
  }).then(async r=>{
    if(r.isConfirmed){
      s.name = r.value.name;
      s.room = r.value.room;
      renderStudentsByRoom();
      saveLocalStorage();
      if(window.updateStudentsCloud) await window.updateStudentsCloud();
      Swal.fire({icon:'success',title:'แก้ไขข้อมูลสำเร็จ',timer:1000,showConfirmButton:false});
    }
  });
}

// Assignments Handler
async function addAssignment(){
  const name = document.getElementById('assignName').value.trim();
  const date = document.getElementById('assignDate').value;
  if(!name || !date){
    Swal.fire({icon:'warning',title:'กรุณากรอกข้อมูลงานและกำหนดส่ง'});
    return;
  }
  window.assignments.push({name, date});
  document.getElementById('assignName').value = '';
  document.getElementById('assignDate').value = '';
  renderAssignments();
  initSubmissionFilters();
  saveLocalStorage();
  if(window.updateAssignmentsCloud) await window.updateAssignmentsCloud();
  Swal.fire({icon:'success',title:'สร้างมอบหมายใบงานใหม่เรียบร้อย',timer:1200,showConfirmButton:false});
}
function renderAssignments(){
  const el = document.getElementById('assignmentList');
  if(!el) return;
  if(window.assignments.length === 0){
    el.innerHTML = `<p class="text-gray-400 text-center py-6">ยังไม่มีมอบหมายใบงานการเรียนการสอน</p>`;
    return;
  }
  el.innerHTML = window.assignments.map((a,idx)=>`
    <div class="flex flex-wrap justify-between items-center bg-white/70 px-4 py-3 rounded-2xl mb-2 border border-gray-100 hover:bg-emerald-50/20 transition-all">
      <div class="flex-1 min-w-[200px]">
        <h4 class="font-bold text-gray-800">${a.name}</h4>
        <p class="text-xs text-gray-500 font-semibold">⏰ กำหนดส่ง: ${new Date(a.date).toLocaleString('th-TH')}</p>
      </div>
      <div class="flex items-center gap-2">
        <button onclick="editAssignment(${idx})" class="text-blue-600 text-xs font-bold hover:underline">แก้ไข</button>
        <button onclick="deleteAssignment(${idx})" class="text-red-500 text-xs font-bold hover:underline">ลบ</button>
      </div>
    </div>
  `).join('');
}
function editAssignment(idx){
  const a = window.assignments[idx];
  Swal.fire({
    title: 'แก้ไขรายละเอียดใบงาน',
    html: `
      <div class="text-left mb-2 text-sm font-semibold text-gray-600">ชื่องาน</div>
      <input id="swal-assign-name" class="swal2-input !mt-1 !mb-4" value="${a.name}">
      <div class="text-left mb-2 text-sm font-semibold text-gray-600">กำหนดส่ง</div>
      <input type="datetime-local" id="swal-assign-date" class="swal2-input !mt-1" value="${a.date}">`,
    showCancelButton: true,
    confirmButtonText: 'บันทึก',
    cancelButtonText: 'ยกเลิก',
    preConfirm: () => {
      const name = document.getElementById('swal-assign-name').value.trim();
      const date = document.getElementById('swal-assign-date').value;
      if(!name || !date) Swal.showValidationMessage('กรุณากรอกข้อมูลให้ครบถ้วน');
      return {name, date};
    }
  }).then(async r=>{
    if(r.isConfirmed){
      if(a.name !== r.value.name) {
        if(window.submissions[a.name]){
          window.submissions[r.value.name] = window.submissions[a.name];
          delete window.submissions[a.name];
        }
      }
      window.assignments[idx] = r.value;
      renderAssignments();
      initSubmissionFilters();
      saveLocalStorage();
      if(window.updateAssignmentsCloud) await window.updateAssignmentsCloud();
      if(window.updateSubmissionsCloud) await window.updateSubmissionsCloud();
      Swal.fire('สำเร็จ', 'แก้ไขข้อมูลใบงานเรียบร้อย', 'success');
    }
  });
}
function deleteAssignment(idx){
  Swal.fire({
    title: 'ต้องการลบใบงานนี้?',
    text: 'การลบจะลบข้อมูลประวัติและการเช็คสถานะงานนี้ทั้งหมด ดำเนินการต่อหรือไม่?',
    icon: 'warning',
    showCancelButton: true,
    confirmButtonText: 'ยืนยัน',
    cancelButtonText: 'ยกเลิก',
    confirmButtonColor: '#dc2626'
  }).then(async r=>{
    if(r.isConfirmed){
      const a = window.assignments[idx];
      if(window.submissions[a.name]) delete window.submissions[a.name];
      window.assignments.splice(idx,1);
      renderAssignments();
      initSubmissionFilters();
      saveLocalStorage();
      if(window.updateAssignmentsCloud) await window.updateAssignmentsCloud();
      if(window.updateSubmissionsCloud) await window.updateSubmissionsCloud();
      Swal.fire('สำเร็จ', 'ลบใบงานที่เลือกเสร็จสิ้น', 'success');
    }
  });
}

// Exams Handler
async function addExam(){
  const name = document.getElementById('examName').value.trim();
  const max = parseFloat(document.getElementById('examMaxScore').value) || 20;
  if(!name){
    Swal.fire({icon:'warning',title:'กรุณากรอกชื่อหัวข้อสอบ'});
    return;
  }
  window.exams.push({name, maxScore: max, scores: {}});
  document.getElementById('examName').value = '';
  renderExams();
  saveLocalStorage();
  if(window.updateExamsCloud) await window.updateExamsCloud();
  Swal.fire({icon:'success',title:'เพิ่มการสอบใหม่เรียบร้อย',timer:1000,showConfirmButton:false});
}
function renderExams(){
  const el = document.getElementById('examList');
  if(!el) return;
  if(window.exams.length === 0){
    el.innerHTML = `<p class="text-gray-400 text-center py-6">ยังไม่มีผลและรายการสอบในระบบ</p>`;
    return;
  }
  let html = '';
  window.exams.forEach((ex,idx)=>{
    html += `
      <div class="mb-4 bg-white/60 p-4 rounded-2xl border border-gray-100">
        <div class="flex justify-between items-center mb-2 flex-wrap gap-2">
          <div>
            <h4 class="font-bold text-gray-800 text-base">${ex.name}</h4>
            <p class="text-xs font-semibold text-emerald-700">คะแนนเต็ม: ${ex.maxScore} คะแนน</p>
          </div>
          <button onclick="deleteExam(${idx})" class="text-red-500 font-bold hover:underline text-xs">ลบผลสอบนี้</button>
        </div>
        <div class="bg-white/80 p-3 rounded-xl max-h-48 overflow-y-auto border border-gray-50 text-xs">
          <div class="font-bold text-gray-600 border-b border-gray-100 pb-1 mb-2 flex justify-between">
            <span>👤 นักเรียน</span>
            <span>คะแนน</span>
          </div>
          ${window.students.map(s => {
            const score = ex.scores[s.id] !== undefined ? ex.scores[s.id] : '-';
            return `
              <div class="flex justify-between items-center py-1 border-b border-gray-50 hover:bg-emerald-50/20">
                <span>${s.name} (${s.id})</span>
                <span class="font-mono font-bold text-emerald-800 cursor-pointer" onclick="editSingleExamScore(${idx}, '${s.id}')">
                  ${score} / ${ex.maxScore} <span class="text-[10px] text-blue-500 font-normal">📝 แก้</span>
                </span>
              </div>
            `;
          }).join('')}
        </div>
      </div>
    `;
  });
  el.innerHTML = html;
  
  const select = document.getElementById('examScoreSelect');
  if(select){
    select.innerHTML = `<option value="">-- เลือกหัวข้อสอบที่จะนำเข้าคะแนน --</option>` + window.exams.map((ex,i)=>`<option value="${i}">${ex.name}</option>`).join('');
  }
}
function deleteExam(idx){
  Swal.fire({
    title: 'ยืนยันลบการสอบ?',
    text: 'การสอบนี้รวมทั้งคะแนนทั้งหมดของนักเรียนจะถูกลบออกจากคลาวด์',
    icon: 'warning',
    showCancelButton: true,
    confirmButtonText: 'ตกลงลบ',
    cancelButtonText: 'ยกเลิก',
    confirmButtonColor: '#dc2626'
  }).then(async r=>{
    if(r.isConfirmed){
      window.exams.splice(idx,1);
      renderExams();
      saveLocalStorage();
      if(window.updateExamsCloud) await window.updateExamsCloud();
      Swal.fire('เรียบร้อย', 'ลบเรียบร้อย', 'success');
    }
  });
}
function editSingleExamScore(examIdx, studentId){
  const ex = window.exams[examIdx];
  const s = window.students.find(x=>x.id===studentId);
  const current = ex.scores[studentId] !== undefined ? ex.scores[studentId] : '';
  Swal.fire({
    title: `ใส่คะแนน: ${s.name}`,
    input: 'number',
    inputValue: current,
    inputAttributes: { min: 0, max: ex.maxScore, step: 0.1 },
    showCancelButton: true,
    confirmButtonText: 'ตกลง',
    cancelButtonText: 'ยกเลิก',
    inputValidator: (val) => {
      if(!val) return 'กรุณาระบุคะแนน';
      const num = parseFloat(val);
      if(num < 0 || num > ex.maxScore) return `คะแนนต้องอยู่ระหว่าง 0 - ${ex.maxScore}`;
    }
  }).then(async res=>{
    if(res.isConfirmed){
      ex.scores[studentId] = parseFloat(res.value);
      renderExams();
      saveLocalStorage();
      if(window.updateExamsCloud) await window.updateExamsCloud();
      Swal.fire({icon:'success',title:'อัปเดตคะแนนแล้ว',timer:1000,showConfirmButton:false});
    }
  });
}
async function importExamScores(){
  const examIdx = document.getElementById('examScoreSelect').value;
  const txt = document.getElementById('importExamScores').value.trim();
  if(!examIdx) { Swal.fire('แจ้งเตือน','กรุณาเลือกหัวข้อการสอบก่อน','warning'); return;}
  if(!txt) { Swal.fire('แจ้งเตือน','กรุณาป้อนข้อความคะแนน','warning'); return;}
  
  const ex = window.exams[examIdx];
  const lines = txt.split('\n').filter(l=>l.trim());
  let count = 0;
  lines.forEach(l=>{
    const parts = l.split(',');
    if(parts.length>=2){
      const id = parts[0].trim();
      const score = parseFloat(parts[1].trim());
      if(window.students.find(s=>s.id===id) && !isNaN(score)){
        ex.scores[id] = score;
        count++;
      }
    }
  });
  document.getElementById('importExamScores').value = '';
  renderExams();
  saveLocalStorage();
  if(window.updateExamsCloud) await window.updateExamsCloud();
  Swal.fire('สำเร็จ', `อัปโหลดคะแนนสอบเรียบร้อย ทั้งหมด ${count} คน`, 'success');
}

// Scan Barcode & Fast Check-In
function updateScanSelect(){
  const sel = document.getElementById('scanAssignment');
  if(!sel) return;
  sel.innerHTML = `<option value="">-- เลือกใบงานมอบหมาย --</option>` + window.assignments.map(a=>`<option value="${a.name}">${a.name}</option>`).join('');
}
async function handleScan(e){
  if(e.key === 'Enter'){
    const val = e.target.value.trim();
    const assignName = document.getElementById('scanAssignment').value;
    if(!assignName){
      Swal.fire('แจ้งเตือน','กรุณาเลือกใบงานที่จะสแกนส่งก่อนการทำงาน','warning');
      e.target.value = '';
      return;
    }
    if(!val) return;
    
    const stu = window.students.find(s => s.id === val);
    if(!stu){
      window.scanLog.unshift(`⚠️ [FAIL] ไม่พบรหัสนักเรียน: ${val} (${new Date().toLocaleTimeString()})`);
      updateScanLogUI();
      e.target.value = '';
      return;
    }
    
    if(!window.submissions[assignName]) window.submissions[assignName] = [];
    if(!window.submissions[assignName].includes(val)){
      window.submissions[assignName].push(val);
    }
    
    window.scanLog.unshift(`✅ [SUCCESS] ${stu.name} (${stu.id}) ห้อง ${stu.room} เช็คชื่อส่งงาน '${assignName}' เรียบร้อย (${new Date().toLocaleTimeString()})`);
    updateScanLogUI();
    saveLocalStorage();
    if(window.updateSubmissionsCloud) await window.updateSubmissionsCloud();
    e.target.value = '';
  }
}
function updateScanLogUI(){
  const logEl = document.getElementById('scanLog');
  if(!logEl) return;
  if(window.scanLog.length===0){
    logEl.innerHTML = '<p class="text-gray-400 text-center py-4">ยังไม่มีประวัติการเช็คชื่อ/สแกนส่งงานในเซสชันนี้</p>';
    return;
  }
  logEl.innerHTML = window.scanLog.map(log => {
    const isSuccess = log.includes('SUCCESS');
    return `<div class="p-2 mb-1 rounded-lg text-xs font-semibold ${isSuccess ? 'bg-green-100 text-green-800':'bg-red-100 text-red-800'}">${log}</div>`;
  }).join('');
}
function clearScanLog(){
  window.scanLog = [];
  updateScanLogUI();
}

// Submissions Filter & Display

  const assignSel = document.getElementById('submissionFilterAssign');
  if(assignSel){
    assignSel.innerHTML = `<option value="">-- งานทั้งหมด --</option>` + window.assignments.map(a=>`<option value="${a.name}">${a.name}</option>`).join('');
  }
  const roomSel = document.getElementById('submissionFilterRoom');
  if(roomSel){
    roomSel.innerHTML = `<option value="">-- ทุกห้องเรียน --</option>` + window.classrooms.map(r=>`<option value="${r}">ม.${r}</option>`).join('');
  }
}
function renderSubmissionStatus(){
  const selectedAssign = document.getElementById('submissionFilterAssign')?.value || '';
  const selectedRoom = document.getElementById('submissionFilterRoom')?.value || '';
  
  let targetStudents = [...window.students];
  if(selectedRoom) targetStudents = targetStudents.filter(s=>s.room===selectedRoom);
  
  let targetAssigns = [...window.assignments];
  if(selectedAssign) targetAssigns = targetAssigns.filter(a=>a.name === selectedAssign);
  
  const el = document.getElementById('submissionTable');
  if(!el) return;
  if(targetStudents.length === 0 || targetAssigns.length === 0){
    el.innerHTML = '<p class="text-gray-400 text-center py-4">ไม่มีข้อมูลที่สอดคล้องตามตัวกรอง</p>';
    return;
  }
  
  let html = `
    <table class="w-full text-xs text-left border-collapse bg-white/80 rounded-2xl overflow-hidden shadow-sm">
      <thead>
        <tr class="bg-emerald-800 text-white font-bold">
          <th class="p-3">รหัส</th>
          <th class="p-3">ชื่อ-สกุล</th>
          <th class="p-3">ห้อง</th>
          ${targetAssigns.map(a=>`<th class="p-3 text-center truncate max-w-[120px]" title="${a.name}">${a.name}</th>`).join('')}
        </tr>
      </thead>
      <tbody>
  `;
  
  targetStudents.forEach(s=>{
    html += `
      <tr class="border-b border-gray-100 hover:bg-emerald-50/30">
        <td class="p-3 font-mono font-bold text-gray-700">${s.id}</td>
        <td class="p-3 font-semibold text-gray-800">${s.name}</td>
        <td class="p-3 font-bold text-emerald-800">ม.${s.room}</td>
        ${targetAssigns.map(a=>{
          const hasSubmitted = window.submissions[a.name] && window.submissions[a.name].includes(s.id);
          const hasFile = window.submittedFiles.some(f => f.assignmentName === a.name && f.studentId === s.id);
          return `
            <td class="p-3 text-center font-bold">
              <span class="cursor-pointer transition-transform inline-block hover:scale-110" onclick="toggleSubmissionState('${a.name}', '${s.id}')">
                ${hasSubmitted ? '🟢 ส่งแล้ว' : '🔴 ยังไม่ส่ง'}
              </span>
              ${hasFile ? '<span class="block text-[10px] text-blue-500 font-normal">📎 มีไฟล์แนบ</span>' : ''}
            </td>
          `;
        }).join('')}
      </tr>
    `;
  });
  
  html += `</tbody></table>`;
  el.innerHTML = html;
}
async function toggleSubmissionState(assignName, studentId){
  if(!window.submissions[assignName]) window.submissions[assignName] = [];
  const idx = window.submissions[assignName].indexOf(studentId);
  if(idx > -1){
    window.submissions[assignName].splice(idx,1);
  } else {
    window.submissions[assignName].push(studentId);
  }
  renderSubmissionStatus();
  saveLocalStorage();
  if(window.updateSubmissionsCloud) await window.updateSubmissionsCloud();
}

function renderSubmittedFiles(){
  const assignFilter = document.getElementById('fileFilterAssign')?.value || '';
  const roomFilter = document.getElementById('fileFilterRoom')?.value || '';
  const searchTxt = (document.getElementById('fileSearchBox')?.value || '').toLowerCase();
  
  const el = document.getElementById('filesList');
  if(!el) return;
  
  let filtered = window.submittedFiles.filter(f => {
    const stu = window.students.find(s => s.id === f.studentId);
    if(!stu) return false;
    
    const matchAssign = !assignFilter || f.assignmentName === assignFilter;
    const matchRoom = !roomFilter || stu.room === roomFilter;
    const matchSearch = !searchTxt || stu.name.toLowerCase().includes(searchTxt) || stu.id.includes(searchTxt);
    
    return matchAssign && matchRoom && matchSearch;
  });
  
  if(filtered.length === 0){
    el.innerHTML = '<p class="text-gray-400 text-center py-6">ไม่พบไฟล์ส่งงานตามเงื่อนไขที่เลือก</p>';
    return;
  }
  
  el.innerHTML = filtered.map((f, i) => {
    const stu = window.students.find(s => s.id === f.studentId);
    return `
      <div class="p-4 mb-2 bg-white/70 rounded-2xl border border-gray-100 flex justify-between items-center flex-wrap gap-2 hover:bg-emerald-50/20 transition-all">
        <div>
          <h4 class="font-bold text-gray-800 text-sm">${f.assignmentName}</h4>
          <p class="text-xs text-gray-500 font-semibold">👤 ส่งโดย: ${stu ? stu.name : 'ไม่ทราบนาม'} (${f.studentId}) ม.${stu ? stu.room : '-'}</p>
          <p class="text-xs text-emerald-700 font-mono font-bold">📄 ${f.fileName} (${f.fileSize}) - ${f.uploadTime}</p>
        </div>
        <div class="flex gap-2">
          ${f.fileBase64 ? `<button onclick="viewFileModal('${f.fileName}', '${f.fileBase64}')" class="btn-info text-white text-xs font-bold px-3 py-2 rounded-xl">ดูไฟล์ภาพ</button>` : ''}
          <button onclick="downloadSimulatedFile('${f.fileName}', '${f.fileBase64 || '#'}')" class="btn-primary text-white text-xs font-bold px-3 py-2 rounded-xl">ดาวน์โหลด</button>
          <button onclick="deleteFileRecord('${f._id}')" class="btn-danger text-white text-xs font-bold px-3 py-2 rounded-xl">ลบไฟล์</button>
        </div>
      </div>
    `;
  }).join('');
}

// ระบบบีบอัดรูปภาพให้มีขนาดพอเหมาะก่อนอัปโหลดขึ้น Firebase
function compressImage(file, callback) {
  // หากเป็น PDF ไม่สามารถย่อขนาดด้วย Canvas ได้ ให้ส่งคืนทันที
  if (file.type === 'application/pdf') {
    const reader = new FileReader();
    reader.onload = (e) => callback(e.target.result);
    reader.readAsDataURL(file);
    return;
  }

  const reader = new FileReader();
  reader.onload = function (event) {
    const img = new Image();
    img.src = event.target.result;
    img.onload = function () {
      const canvas = document.createElement('canvas');
      const ctx = canvas.getContext('2d');

      // กำหนดขนาดกว้างสูงสุด 800px โดยคงอัตราส่วนรูปเดิมไว้
      const MAX_WIDTH = 800;
      let width = img.width;
      let height = img.height;

      if (width > MAX_WIDTH) {
        height *= MAX_WIDTH / width;
        width = MAX_WIDTH;
      }

      canvas.width = width;
      canvas.height = height;
      ctx.drawImage(img, 0, 0, width, height);

      // แปลงข้อมูลและบีบอัดความละเอียดภาพเหลือ 60% (.jpg / .webp)
      const dataUrl = canvas.toDataURL('image/jpeg', 0.6);
      callback(dataUrl);
    };
  };
  reader.readAsDataURL(file);
}

// File Submit Handler
function studentSubmit(){
  if(!window.currentUser || window.currentUser.role !== 'student') return;
  const s = window.currentUser.data;
  const assignName = document.getElementById('stuSubmitAssign').value;
  const fileInput = document.getElementById('stuSubmitFile');
  const file = fileInput.files[0];
  
  if(!assignName){
    Swal.fire({icon:'warning',title:'กรุณาเลือกใบงานก่อนส่ง'});
    return;
  }
  if(!file){
    Swal.fire({icon:'warning',title:'กรุณาเลือกไฟล์งานแนบเพื่อดำเนินการ'});
    return;
  }
  
  Swal.fire({
    title: 'กำลังเตรียมไฟล์และส่งงาน...',
    text: 'กรุณารอสักครู่ ระบบกำลังบีบอัดภาพและอัปโหลดขึ้นเซิร์ฟเวอร์คลาวด์',
    allowOutsideClick: false,
    didOpen: () => { Swal.showLoading(); }
  });

  // ใช้การบีบอัดขนาดภาพเพื่อหลีกเลี่ยงข้อผิดพลาดเรื่องขนาดไฟล์เกินลิมิตคลาวด์
  compressImage(file, async function(base64) {
    try {
      if(!window.submissions[assignName]) window.submissions[assignName] = [];
      if(!window.submissions[assignName].includes(s.id)){
        window.submissions[assignName].push(s.id);
      }
      
      // คำนวณขนาดหลังการบีบอัดจำลอง
      const compressedSizeKB = Math.round((base64.length * 3) / 4 / 1024);
      const fileSizeStr = compressedSizeKB + ' KB';
      
      // ลบข้อมูลไฟล์เก่าที่เคยส่งในใบงานเดียวกันนี้ออกก่อน
      window.submittedFiles = window.submittedFiles.filter(f => !(f.assignmentName === assignName && f.studentId === s.id));
      
      // บันทึกการส่งงานลงเซสชันสำรองก่อนป้องกันกรณีเน็ตหลุด
      window.submittedFiles.push({
        assignmentName: assignName,
        studentId: s.id,
        fileName: file.name,
        uploadTime: new Date().toLocaleString('th-TH'),
        fileSize: fileSizeStr,
        fileBase64: base64
      });
      saveLocalStorage();

      // บันทึกขึ้นระบบคลาวด์กลาง
      if(window.uploadFileCloud) {
        await window.uploadFileCloud(assignName, s.id, file.name, fileSizeStr, base64);
      }
      if(window.updateSubmissionsCloud) {
        await window.updateSubmissionsCloud();
      }
      
      renderStudentPage();
      fileInput.value = '';
      document.getElementById('stuFileStatus').textContent = 'ยังไม่ได้เลือกไฟล์';
      
      Swal.fire({
        icon:'success',
        title:'ส่งใบงานสำเร็จเรียบร้อย',
        text:`ไฟล์ภาพได้รับการย่อขนาดให้พอดีและอัปโหลดแล้ว คุณครูสามารถเปิดตรวจได้ทันที`,
        timer:2000
      });
    } catch (err) {
      console.error("Upload process failed:", err);
      // หากคลาวด์หลักขัดข้อง แต่อัปโหลดเสร็จบน LocalStorage แล้ว จะแจ้งเตือนให้สบายใจ
      Swal.fire({
        icon:'warning',
        title:'บันทึกข้อมูลแบบออฟไลน์แล้ว',
        text:'เซิร์ฟเวอร์คลาวด์ตอบสนองช้า ระบบบันทึกงานไว้ที่เครื่องคุณครูชั่วคราวแล้วกรุณาอย่าล้างข้อมูลเบราว์เซอร์',
        confirmButtonText: 'ตกลง'
      });
    }
  });
}
